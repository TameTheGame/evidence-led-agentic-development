## Cause

`Code/OutpostGenerator.cs:48` created `MeshComponent` enabled, then assigned `Mesh`. In a game scene the component builds its model and collision at enable time, so every mesh block came out with no collision. The apron is the only walkable floor built that way, so the gate is where it shows as falling — the walls and the tower lost their collision too, they just read as "walked through a wall" rather than "fell".

Fixed by creating disabled → assigning → enabling, and added `UnitTests/CollisionTests.cs` with `OUT_COLL_01_EveryFloorHasCollision`, the check the spec already names for that requirement. It traces straight down onto the middle of each walkable floor's top face (the yard floor and anything within step height of it — that's the gate apron) and asserts a player lands on it at the z the layout claims.

+using System;

+/// <summary>
+/// OUT-COLL-01 — every walkable floor in the layout has collision.
+///
+/// If a run fails with a NullReferenceException from ModelBuilder.AddSurface, the engine had
+/// no "default" physics surface loaded: headless test runs don't mount base content, so mesh

+/// </summary>
+[TestClass]
+public class CollisionTests
+{
+	/// <summary>
+	/// How far above the yard floor a top face can sit and still be something a walking player
+	/// steps onto rather than an obstacle. The gate apron is a floor by this rule; the walls and
+	/// the tower are not.
+	/// </summary>
+	const float StepHeight = 18f;
+
+	[TestMethod]
+	public void OUT_COLL_01_EveryFloorHasCollision()
+	{
+		var layout = OutpostLayout.Parse( File.ReadAllText( LayoutPath() ) );
+
+		var scene = new Scene();
+		using var scope = scene.Push();
+		OutpostGenerator.Build( scene, layout );
+
+		var floors = WalkableFloors( layout );
+		Assert.AreNotEqual( 0, floors.Count, "found no walkable floors in the layout; the check would pass blind" );
+
+		foreach ( var floor in floors )
+		{
+			// Straight down onto the middle of the top face, starting clear of it and ending
+			// just below it. The spawn and route checks cover the other places a player stands.
+			var from = floor.Top + Vector3.Up * 32f;
+			var to = floor.Top - Vector3.Up * 8f;
+			var trace = scene.Trace.Ray( from, to ).Run();
+
+			Assert.IsTrue( trace.Hit,
+				$"{floor.Id}: nothing to stand on at {floor.Top} — a player here falls through" );
+			Assert.IsTrue( MathF.Abs( trace.HitPosition.z - floor.Top.z ) <= 1f,
+				$"{floor.Id}: stood on z={trace.HitPosition.z}, layout says the top is z={floor.Top.z}" );
+		}
+	}
+
+	record Floor( string Id, Vector3 Top );
+
+	/// <summary>
+	/// The yard floor, plus every block low enough to walk onto, each with the middle of its top
+	/// face in world space.
+	/// </summary>
+	static List<Floor> WalkableFloors( OutpostLayout layout )
+	{
+		var groundTop = layout.Ground.Max[2];
+		var floors = new List<Floor>
+		{
+			new( "ground", new Vector3(
+				(layout.Ground.Min[0] + layout.Ground.Max[0]) / 2,
+				(layout.Ground.Min[1] + layout.Ground.Max[1]) / 2,

+		foreach ( var block in layout.Blocks )

+			var top = block.Min[2] + block.Size[2];
+			if ( top > groundTop + StepHeight )

+			floors.Add( new( block.Id, new Vector3(
+				block.Min[0] + block.Size[0] / 2,
+				block.Min[1] + block.Size[1] / 2,
+				top ) ) );
+		}
