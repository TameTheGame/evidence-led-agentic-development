+using System;

+using Microsoft.VisualStudio.TestTools.UnitTesting;

+using Assert = Microsoft.VisualStudio.TestTools.UnitTesting.Assert;

+/// <summary>
+/// Engine-rung collision checks. Each one builds the real outpost from the layout data into
+/// a headless scene and traces against it, because rendered geometry is not collision.
+///
+/// OUT_COLL_01 leans on the ground, which is a box collider and is known to work in game, so
+/// it doubles as a control: if OUT_COLL_01 fails too, suspect the harness rather than the
+/// blocks (see the note in OUT_COLL_02).
+/// </summary>
+[TestClass]
+public class CollisionTests
+{
+	/// <summary>How far below a walkable point a floor may be before a player would fall.</summary>
+	const float FloorReach = 32f;
+
+	/// <summary>How far outside a block's face a probe ray starts and ends.</summary>
+	const float ProbeMargin = 64f;
+
+	/// <summary>OUT-COLL-01 — every walkable floor in the layout has collision.</summary>
+	[TestMethod]
+	public void OUT_COLL_01_EveryFloorHasCollision()
+	{
+		var layout = Layout();
+		var scene = new Scene();
+		using var scope = scene.Push();
+		OutpostGenerator.Build( scene, layout );
+
+		foreach ( var (id, point) in WalkablePoints( layout ) )
+		{
+			var from = point + Vector3.Up * 8;
+			var to = point + Vector3.Down * FloorReach;
+			var trace = scene.Trace.Ray( from, to ).Run();
+
+			Assert.IsTrue( trace.Hit,
+				$"{id}: nothing solid within {FloorReach} units below {point}, so a player there has no floor" );
+		}
+	}
+
+	/// <summary>
+	/// OUT-COLL-02 — every block in the layout is solid. A ray along the block's thinnest
+	/// axis, through its centre, must hit something. This is what catches a wall that renders
+	/// but does not block.
+	/// </summary>
+	[TestMethod]
+	public void OUT_COLL_02_EveryBlockIsSolid()
+	{
+		var layout = Layout();
+		var scene = new Scene();
+		using var scope = scene.Push();
+		OutpostGenerator.Build( scene, layout );
+
+		foreach ( var block in layout.Blocks )
+		{
+			var centre = new[]
+			{
+				block.Min[0] + block.Size[0] / 2,
+				block.Min[1] + block.Size[1] / 2,
+				block.Min[2] + block.Size[2] / 2,
+			};
+
+			// Probe across the block the short way, so the ray cannot slip past a corner.
+			var axis = 0;
+			for ( var i = 1; i < 3; i++ )
+				if ( block.Size[i] < block.Size[axis] )
+					axis = i;
+
+			var reach = block.Size[axis] / 2 + ProbeMargin;
+			var from = (float[])centre.Clone();
+			var to = (float[])centre.Clone();
+			from[axis] -= reach;
+			to[axis] += reach;
+
+			var trace = scene.Trace.Ray( V( from ), V( to ) ).Run();
+
+			Assert.IsTrue( trace.Hit,
+				$"{block.Id}: a ray through its centre along axis {axis} hit nothing, " +
+				$"so the block is not solid and a player walks through it" );
+		}
+	}
