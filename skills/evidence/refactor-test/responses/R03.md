## Final reply

Done. The mesh-block code now lives in its own class.

**`Code/MeshBlock.cs`** (new) — `public static class MeshBlock` in `namespace Frontier`:
- `Create( Scene, string id, Vector3 min, Vector3 size )` — the old `AddMeshBlock`, now public
- `BoxMesh( Vector3 size )` — private, unchanged geometry

I carried the disabled-create/assign/enable comment from `UnitTests/StarterGeometry.cs` onto `Create`, since that ordering is the non-obvious part of the method and the original had it only in the test copy.

**`Code/AreaGenerator.cs`** — both methods removed; the loop in `Build` now calls `MeshBlock.Create`. The generator keeps the ground collider, the `V` helper, and `AreaLayout`.

+using Sandbox;
+
+namespace Frontier;
+
+/// <summary>
+/// A static box of generated mesh geometry with mesh collision. Walls, cliffs, aprons, docks,
+/// and landmarks are all built from these.
+/// </summary>
+public static class MeshBlock
+{
+	/// <summary>
+	/// Creates a block in the scene. <paramref name="min"/> is its lowest corner in world space.
+	/// </summary>
+	public static MeshComponent Create( Scene scene, string id, Vector3 min, Vector3 size )
+	{
+		var go = scene.CreateObject();
+		go.Name = id;
+		go.WorldPosition = min;
+
+		// Outside an editor scene, MeshComponent builds its model only when it is enabled;
+		// setting Mesh afterwards does nothing. Create it disabled, assign the mesh, then enable.
+		var block = go.Components.Create<MeshComponent>( false );
+		block.Collision = MeshComponent.CollisionType.Mesh;
+		block.Mesh = BoxMesh( size );
+		block.Enabled = true;
+
+		return block;
+	}
+
+	/// <summary>
+	/// An axis-aligned box mesh with one corner at the local origin.
+	/// </summary>
+	static PolygonMesh BoxMesh( Vector3 size )
+	{
+		var mesh = new PolygonMesh();
+		var v = mesh.AddVertices(
+			new Vector3( 0, 0, 0 ), new Vector3( size.x, 0, 0 ),
+			new Vector3( size.x, size.y, 0 ), new Vector3( 0, size.y, 0 ),
+			new Vector3( 0, 0, size.z ), new Vector3( size.x, 0, size.z ),
+			new Vector3( size.x, size.y, size.z ), new Vector3( 0, size.y, size.z ) );
+
+		mesh.AddFace( v[0], v[3], v[2], v[1] ); // bottom
+		mesh.AddFace( v[4], v[5], v[6], v[7] ); // top
+		mesh.AddFace( v[0], v[1], v[5], v[4] );
+		mesh.AddFace( v[1], v[2], v[6], v[5] );
+		mesh.AddFace( v[2], v[3], v[7], v[6] );
+		mesh.AddFace( v[3], v[0], v[4], v[7] );
+
+		return mesh;
+	}
+}
