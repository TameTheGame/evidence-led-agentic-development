Rigor: light — a mechanical move with no behavior change, ordinary revert, and the existing outpost/quarry collision and spawn tests exercise the moved code through `AreaGenerator.Build`.

- `UnitTests/StarterGeometry.cs` still has its own copy of the same box-mesh and block code. It's starter scaffolding that deliberately doesn't depend on the game code, so folding it into `MeshBlock` is a judgment call about whether the starter tests should stay self-contained.

+using Sandbox;
+
+namespace Frontier;
+
+/// <summary>
+/// A static box of generated mesh geometry with mesh collision. Walls, cliffs, aprons,
+/// docks, and landmarks in an area's layout are all mesh blocks.
+/// </summary>
+public static class MeshBlock
+{
+	/// <summary>
+	/// Adds a mesh block to the scene. <paramref name="min"/> is its lowest corner in world space.
+	/// </summary>
+	public static MeshComponent Create( Scene scene, string id, Vector3 min, Vector3 size )
+	{
+		var go = scene.CreateObject();
+		go.Name = id;
+		go.WorldPosition = min;
+
+		// Outside an editor scene, MeshComponent builds only when it is enabled; setting Mesh
+		// afterwards does nothing. Create it disabled, assign the mesh, then enable it.
+		var block = go.Components.Create<MeshComponent>( false );
+		block.Collision = MeshComponent.CollisionType.Mesh;
+		block.Mesh = BoxMesh( size );
+		block.Enabled = true;
+
+		return block;
+	}
+
+	/// <summary>
+	/// An axis-aligned box mesh with one corner at the local origin.
+	/// </summary>
+	static PolygonMesh BoxMesh( Vector3 size )
+	{
+		var mesh = new PolygonMesh();
+		var v = mesh.AddVertices(
+			new Vector3( 0, 0, 0 ), new Vector3( size.x, 0, 0 ),
+			new Vector3( size.x, size.y, 0 ), new Vector3( 0, size.y, 0 ),
+			new Vector3( 0, 0, size.z ), new Vector3( size.x, 0, size.z ),
+			new Vector3( size.x, size.y, size.z ), new Vector3( 0, size.y, size.z ) );
+
+		mesh.AddFace( v[0], v[3], v[2], v[1] ); // bottom
+		mesh.AddFace( v[4], v[5], v[6], v[7] ); // top
+		mesh.AddFace( v[0], v[1], v[5], v[4] );
+		mesh.AddFace( v[1], v[2], v[6], v[5] );
+		mesh.AddFace( v[2], v[3], v[7], v[6] );
+		mesh.AddFace( v[3], v[0], v[4], v[7] );
+
+		return mesh;
+	}
+}
