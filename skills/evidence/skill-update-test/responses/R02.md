## Final reply

Written: `UnitTests/OutpostSceneTests.cs` — one test, `NorthSpawnPlayerLandsOnSceneGround`.

**What it does.** Loads `Assets/scenes/outpost.scene` through the existing `EngineTestHelpers.LoadSavedScene` (which pulls in `outpost.scene_d`, so the ground mesh isn't empty), reads the spawn position out of the loaded scene, and asserts four things: the scene loaded; the player isn't buried at the spawn; it becomes grounded within 120 ticks; and the surface underfoot is the scene's own `ground` object, not a wall top, the apron, or the tower.

**Two things drove the design.** The README says you reshaped the ground by hand after generating, so the test never assumes the layout's `z = 0` ground top — it takes the spawn from the scene and identifies what the player rests on by object identity instead of by height. And the saved ground is a `MeshComponent` with mesh collision, not the generator's `BoxCollider`, so the downward trace is written to hit the mesh's top face from above, the way mesh collision actually behaves.

+using System.Linq;
+using Microsoft.VisualStudio.TestTools.UnitTesting;
+using Sandbox;
+
+namespace EngineTests;

+/// <summary>
+/// Engine-rung checks against the saved scene, Assets/scenes/outpost.scene, rather than
+/// against geometry built by the generator. The owner generated that scene, reshaped its
+/// ground by hand in the Editor, and saved it, so its ground is an editable mesh whose
+/// height is whatever they saved — not the flat z = 0 top the layout data describes. These
+/// checks therefore read their positions out of the loaded scene and never assume a ground
+/// height.
+/// </summary>
+[TestClass]
+public class OutpostSceneTests
+{
+	const string ScenePath = "Assets/scenes/outpost.scene";
+
+	/// <summary>
+	/// A player placed at the scene's north spawn is not buried in geometry, and falls and
+	/// settles on the scene's own ground object.
+	/// </summary>
+	[TestMethod]
+	public void NorthSpawnPlayerLandsOnSceneGround()
+	{
+		var scene = new Scene();
+
+		using ( scene.Push() )
+		{
+			Assert.IsTrue( EngineTestHelpers.LoadSavedScene( scene, ScenePath ),
+				$"{ScenePath} did not load, so nothing below was tested. Read logs\\testhost.log in the S&box install folder: load errors land there, not in the test output." );
+
+			// The scene is the thing under test, so take the spawn from it rather than from the
+			// layout data. Whether the two still agree is a separate, static-rung claim.
+			var spawn = Find( scene, "spawn.north" );
+			Assert.IsNotNull( spawn, $"{ScenePath} has no object named \"spawn.north\"." );
+
+			// The ground is one object in this scene. Holding on to it lets the landing check
+			// below prove the player came to rest on the ground itself and not on a wall top,
+			// the gate apron, or the tower.
+			var ground = Find( scene, "ground" );
+			Assert.IsNotNull( ground, $"{ScenePath} has no object named \"ground\"." );
+			Assert.IsNotNull( ground.Components.Get<MeshComponent>(),
+				"The scene's \"ground\" object has no mesh, so there is no ground to land on." );
+
+			var pc = EngineTestHelpers.CreatePlayer( scene, spawn.WorldPosition );
+
+			// Check this before ticking. If the hand-reshaped ground rose above the spawn, the
+			// player starts inside it, and after landing a resting body looks much the same.
+			Assert.IsFalse( EngineTestHelpers.BodyStartsInsideGeometry( pc ),
+				$"A player at the north spawn {spawn.WorldPosition} starts inside geometry. The saved ground may have been reshaped up past the spawn." );
+
+			// 120 ticks is about 2.4s at this scene's 50Hz fixed update: far longer than the
+			// short drop from the spawn needs, with room for a ground reshaped downwards.
+			Assert.IsTrue( EngineTestHelpers.TickUntilGrounded( scene, pc, maxTicks: 120 ),
+				$"A player at the north spawn {spawn.WorldPosition} never reached the ground, and ended at {pc.GameObject.WorldPosition}." );
+
+			// Mesh collision is a hollow surface, so this hits the ground mesh's top face from
+			// above, the same way the player's own ground check does. Starting 2 units up clears
+			// resting contact; 8 units is far enough to find the surface underfoot and too short
+			// to punch through to anything further down.
+			var feet = pc.GameObject.WorldPosition + Vector3.Up * 2;
+			var under = pc.Scene.Trace.Ray( feet, feet + Vector3.Down * 8 )
+				.IgnoreGameObjectHierarchy( pc.GameObject )
+				.WithCollisionRules( pc.Tags )
+				.Run();
+
+			Assert.IsTrue( under.Hit, $"The player reports standing on the ground, but nothing was found underfoot at {pc.GameObject.WorldPosition}." );
+			Assert.AreSame( ground, under.GameObject,
+				$"The player landed on \"{under.GameObject?.Name}\" at {pc.GameObject.WorldPosition}, not on the scene's ground." );
+		}
+	}
+
+	/// <summary>
+	/// The first object in the scene with this name, at any depth. The saved scene keeps its
+	/// objects at the root, but searching the whole tree means grouping them in the Editor
+	/// later won't quietly turn this into a null.
+	/// </summary>
+	static GameObject Find( GameObject root, string name ) =>
+		root.Children.FirstOrDefault( child => child.Name == name )
+			?? root.Children.Select( child => Find( child, name ) ).FirstOrDefault( found => found is not null );
