## What the test does

`UnitTests/OutpostSceneTests.cs` → `NorthSpawnLandsOnSavedSceneGround`:

1. Loads `Assets/scenes/outpost.scene` with `EngineTestHelpers.LoadSavedScene`, so the hand-reshaped ground in `outpost.scene_d` comes with it (the public `LoadFromJson` route would drop it and the player would fall through).
2. Takes the spawn and the ground **from the scene**, by name — `spawn.north` (asserting it really carries a `SpawnPoint`) and `ground` — not from `spec/outpost.layout.json`, since the claim is about the scene the game plays.
3. Asserts the player isn't already stuck in geometry before ticking, otherwise `IsOnGround` could be true without anything having been landed on.
4. Ticks up to 240 frames for it to settle. The ground was reshaped by hand, so the drop isn't knowable from the layout; the failure message prints the end position, which separates "fell a long way" (empty mesh or a hole) from "never moved" (scene still loading).
5. Traces down from its feet and asserts the thing under it is the `ground` object — not a wall or the tower.

I deliberately did **not** name it `OUT_SPAWN_01_EverySpawnLandsOnSolidGround`. That requirement is about every spawn in the *generated* outpost; this is one spawn in the *saved* scene, which is a different claim. Naming it that would have made `spec_lint` report OUT-SPAWN-01 as covered when it isn't.

+using System.Linq;
+using Microsoft.VisualStudio.TestTools.UnitTesting;
+using Sandbox;
+
+namespace EngineTests;
+
+/// <summary>
+/// Engine-rung checks against the saved outpost scene, as the owner saved it from the Editor.
+/// These load Assets/scenes/outpost.scene itself, so they see the hand-reshaped ground whose
+/// mesh data lives in outpost.scene_d, not the geometry the generator would build.
+/// The spec's OUT-SPAWN-01 covers spawns in the generated outpost; this is its saved-scene
+/// counterpart and is not an agreed requirement yet.
+/// </summary>
+[TestClass]
+public class OutpostSceneTests
+{
+	const string ScenePath = "Assets/scenes/outpost.scene";
+
+	/// <summary>
+	/// A player put at the scene's north spawn point settles standing on the scene's "ground"
+	/// object: not already stuck in geometry, not fallen through, and not resting on a wall or
+	/// the tower.
+	/// </summary>
+	[TestMethod]
+	public void NorthSpawnLandsOnSavedSceneGround()
+	{
+		var scene = new Scene();
+		using var active = scene.Push();
+
+		Assert.IsTrue( EngineTestHelpers.LoadSavedScene( scene, ScenePath ), $"{ScenePath} did not load" );
+
+		// The spawn and the ground both come from the scene, because the claim is about the
+		// scene the game actually plays, not about the layout data it was generated from.
+		var spawn = scene.Children.FirstOrDefault( o => o.Name == "spawn.north" );
+		Assert.IsNotNull( spawn, "the scene has no object named 'spawn.north'" );
+		Assert.IsNotNull( spawn.Components.Get<SpawnPoint>(), "'spawn.north' has no SpawnPoint component" );
+
+		var ground = scene.Children.FirstOrDefault( o => o.Name == "ground" );
+		Assert.IsNotNull( ground, "the scene has no object named 'ground'" );
+
+		var pc = EngineTestHelpers.CreatePlayer( scene, spawn.WorldPosition );
+
+		// A player that starts stuck inside geometry can report IsOnGround without having
+		// landed on anything, so rule that out before ticking.
+		Assert.IsFalse( EngineTestHelpers.BodyStartsInsideGeometry( pc ),
+			$"the player starts inside geometry at the north spawn {spawn.WorldPosition}" );
+
+		// Generous but bounded: the ground was reshaped by hand, so the drop isn't knowable
+		// from the layout. 240 ticks is several thousand units of fall.
+		var landed = EngineTestHelpers.TickUntilGrounded( scene, pc, 240 );
+		Assert.IsTrue( landed,
+			$"the player never landed; it ended at {pc.GameObject.WorldPosition}, from the north spawn " +
+			$"{spawn.WorldPosition}. A long fall means the ground mesh loaded empty or has a hole at the " +
+			"spawn; no movement at all means the scene never finished loading." );
+
+		// IsOnGround doesn't say what it landed on, so name the surface under its feet.
+		var feet = pc.GameObject.WorldPosition;
+		var under = scene.Trace.Ray( feet + Vector3.Up * 2, feet + Vector3.Down * 16 )
+			.IgnoreGameObjectHierarchy( pc.GameObject )
+			.WithCollisionRules( pc.Tags )
+			.Run();
+
+		Assert.IsTrue( under.Hit,
+			$"nothing is under the player's feet at {feet}, though it reports standing on the ground" );
+		Assert.AreSame( ground, under.GameObject,
+			$"the player landed on '{under.GameObject?.Name}' at {feet}, not on the scene's ground" );
+	}
+}
