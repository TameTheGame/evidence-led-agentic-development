## Final reply

Written, and the static checks I can run here pass. Here's where it stands.

**What the test does** — `UnitTests/OutpostSceneTests.cs`:

1. Loads `Assets/scenes/outpost.scene` through the starter's `EngineTestHelpers.LoadSavedScene`, which also reads the `outpost.scene_d` mesh data (without it the hand-reshaped ground loads empty and a player falls straight through).
2. Reads the north spawn's position from the scene's own `spawn.north` object, puts a headless `PlayerController` there, and ticks until it is grounded.
3. A second test is the negative control: same scene, but with the `ground` object disabled, the landing check must go red. Nothing else in the scene sits under `(0, 600)`, so if that one passes too, the first test really saw the ground and isn't just reporting a player that counted as grounded without ever falling.

I wrote the control as a test rather than doing it by hand because I can't run either one — there's no S&box install here, so the owner's first `dotnet test` is the first run of both. That's the only way to get the "watch it fail for the right reason" evidence in the same unattended run.

+using System;
+using System.Linq;
+using Microsoft.VisualStudio.TestTools.UnitTesting;
+using Sandbox;
+
+namespace EngineTests;
+
+/// <summary>
+/// Engine-rung checks on the saved outpost scene, the one the game loads. The owner reshaped
+/// its ground by hand after generating it, so neither the layout file nor the generator can say
+/// what a player standing in the yard rests on: only the saved scene and the mesh data in
+/// "outpost.scene_d" beside it can, which is why this runs on the engine rung.
+/// </summary>
+[TestClass]
+public class OutpostSceneTests
+{
+	const string ScenePath = "Assets/scenes/outpost.scene";
+	const string GroundName = "ground";
+	const string NorthSpawnName = "spawn.north";
+
+	Scene _scene;
+	IDisposable _scope;
+
+	[TestInitialize]
+	public void LoadScene()
+	{
+		_scene = new Scene();
+		_scope = _scene.Push();
+
+		Assert.IsTrue( EngineTestHelpers.LoadSavedScene( _scene, ScenePath ),
+			$"{ScenePath} did not load" );
+	}
+
+	[TestCleanup]
+	public void UnloadScene()
+	{
+		_scope?.Dispose();
+		_scene?.Destroy();
+	}
+
+	/// <summary>
+	/// OUT-SPAWN-02 (draft): a player put at the north spawn falls and comes to rest on the
+	/// saved ground. The spawn position comes from the scene's own "spawn.north" object, so
+	/// moving that spawn in the Editor moves the check with it.
+	/// </summary>
+	[TestMethod]
+	public void OUT_SPAWN_02_NorthSpawnLandsOnSavedGround()
+	{
+		var spawn = FindObject( NorthSpawnName ).WorldPosition;
+		var player = EngineTestHelpers.CreatePlayer( _scene, spawn );
+
+		Assert.IsTrue( EngineTestHelpers.TickUntilGrounded( _scene, player ),
+			$"a player put at {NorthSpawnName} {spawn} never came to rest on the ground" );
+	}
+
+	/// <summary>
+	/// The negative control for the check above. With the ground disabled, nothing else in the
+	/// scene is under the north spawn, so the same landing check has to go red; if it doesn't,
+	/// the player is resting on something that isn't the ground, or is counted as grounded
+	/// without ever falling, and the check above proves nothing. It runs as a test rather than
+	/// as a note because this environment has no S&box install, so the owner's first run is the
+	/// first run of either check.
+	/// </summary>
+	[TestMethod]
+	public void OUT_SPAWN_02_NegativeControl_NoLandingWithoutTheGround()
+	{
+		FindObject( GroundName ).Enabled = false;
+
+		var spawn = FindObject( NorthSpawnName ).WorldPosition;
+		var player = EngineTestHelpers.CreatePlayer( _scene, spawn );
+
+		Assert.IsFalse( EngineTestHelpers.TickUntilGrounded( _scene, player ),
+			$"with \"{GroundName}\" disabled, a player at {NorthSpawnName} {spawn} still counted "
+			+ "as grounded, so landing there does not show it lands on the ground" );
+	}
+
+	/// <summary>
+	/// A top-level object of the saved scene, by name.
+	/// </summary>
+	GameObject FindObject( string name )
+	{
+		var go = _scene.Children.FirstOrDefault( x => x.Name == name );
+		Assert.IsNotNull( go, $"the saved scene has no object named \"{name}\"" );
+		return go;
+	}
+}
