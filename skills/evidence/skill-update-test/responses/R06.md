## What I wrote

`UnitTests/OutpostSpawnTests.cs` — `OUT_SPAWN_02_SavedSceneNorthSpawnLandsOnGround`. It loads the scene with the starter's `LoadSavedScene` (which reads `outpost.scene_d`), finds the scene's own `spawn.north` and `ground` objects by name, and then asserts in this order:

1. **Something solid is under the spawn, and it's `ground`** — traced *before* any ticking. This is the guard against the known `_d`-data hazard: if the mesh data were dropped, the ground would build empty and traces would pass through, which would otherwise look identical to "the player never landed".
2. **The player doesn't start inside geometry** — via `BodyStartsInsideGeometry`. A body spawned inside the slab can read as grounded, so without this the landing check could pass for the wrong reason.
3. **It lands** within 120 ticks.
4. **It landed on `ground`** — re-traced from where it came to rest, because `IsOnGround` is just as true on a wall top or the tower.

It deliberately asserts **no landing height**. Your README says you reshaped the ground by hand after generating, so the layout's `ground.max.z = 0` is no longer trustworthy for the saved scene — identity of the object underfoot is the thing that's actually knowable.

I also added `OUT-SPAWN-02` to the spec as **`draft`**. `OUT-SPAWN-01` is agreed against the layout and generator code; the saved scene was hand-edited after generation, so it's a separate claim and the generated scene passing wouldn't cover it. Only you can mark it agreed.

+using System.Linq;
+using Microsoft.VisualStudio.TestTools.UnitTesting;
+using Sandbox;
+
+namespace EngineTests;
+
+/// <summary>
+/// Engine-rung checks against the saved outpost scene, as opposed to a scene the generator
+/// builds. The owner reshaped the saved ground by hand, so nothing here assumes the ground
+/// height in spec/outpost.layout.json; the scene's own "ground" object is the subject.
+/// </summary>
+[TestClass]
+public class OutpostSpawnTests
+{
+	const string ScenePath = "Assets/scenes/outpost.scene";
+
+	/// <summary>
+	/// OUT-SPAWN-02 (draft): a player placed at the saved scene's north spawn falls and comes to
+	/// rest on the scene's own ground.
+	/// </summary>
+	[TestMethod]
+	public void OUT_SPAWN_02_SavedSceneNorthSpawnLandsOnGround()
+	{
+		using var scene = new Scene();
+		using ( scene.Push() )
+		{
+			Assert.IsTrue( EngineTestHelpers.LoadSavedScene( scene, ScenePath ),
+				$"{ScenePath} did not load. Engine errors go to logs\\testhost.log in the S&box install folder, not to this output." );
+
+			var ground = Find( scene, "ground" );
+			var spawn = Find( scene, "spawn.north" );
+			var player = EngineTestHelpers.CreatePlayer( scene, spawn.WorldPosition );
+
+			// Before ticking: is there anything solid under the spawn to land on? The scene's
+			// editable meshes keep their geometry in outpost.scene_d, and a load route that drops
+			// that data builds them empty, so traces pass straight through. Asking first keeps that
+			// failure from reading as "the player never landed".
+			var below = TraceDown( player, spawn.WorldPosition );
+			Assert.IsTrue( below.Hit,
+				$"nothing solid under the north spawn at {spawn.WorldPosition}: the saved scene's mesh collision is missing or built empty." );
+			Assert.AreSame( ground, below.GameObject,
+				$"the first solid thing under the north spawn is \"{below.GameObject?.Name}\", not the scene's ground." );
+
+			// A body spawned inside the ground slab can read as grounded, which would make the
+			// landing check pass for the wrong reason.
+			Assert.IsFalse( EngineTestHelpers.BodyStartsInsideGeometry( player ),
+				$"the player starts inside geometry at the north spawn ({spawn.WorldPosition}), so landing there would prove nothing." );
+
+			Assert.IsTrue( EngineTestHelpers.TickUntilGrounded( scene, player, 120 ),
+				$"the player never landed within 120 ticks: it fell from {spawn.WorldPosition} to {player.GameObject.WorldPosition}." );
+
+			// Landed on what? IsOnGround is just as true on a wall top or the tower.
+			var landedOn = TraceDown( player, player.GameObject.WorldPosition );
+			Assert.AreSame( ground, landedOn.GameObject,
+				$"the player landed on \"{landedOn.GameObject?.Name}\" at {player.GameObject.WorldPosition}, not on the scene's ground." );
+		}
+	}
+
+	/// <summary>
+	/// The first solid thing below <paramref name="from"/>, ignoring the player's own body. It
+	/// starts 2 units up so a body already resting on the floor still traces down to it.
+	/// </summary>
+	static SceneTraceResult TraceDown( PlayerController player, Vector3 from ) =>
+		player.Scene.Trace.Ray( from + Vector3.Up * 2, from - Vector3.Up * 16384 )
+			.IgnoreGameObjectHierarchy( player.GameObject )
+			.WithCollisionRules( player.Tags )
+			.Run();
+
+	static GameObject Find( Scene scene, string name )
+	{
+		var go = scene.GetAllObjects( true ).FirstOrDefault( o => o.Name == name );
+		Assert.IsNotNull( go, $"{ScenePath} has no enabled object named \"{name}\"." );
+		return go;
+	}
+}
