+using System.Linq;
+using Microsoft.VisualStudio.TestTools.UnitTesting;
+using Sandbox;
+
+namespace EngineTests;
+
+/// <summary>
+/// Tests against the saved outpost scene, the one the game loads. The owner reshaped its
+/// ground by hand after generating it, and the Editor keeps that shape in
+/// "outpost.scene_d" beside the scene, so a test of the generator says nothing about what
+/// is actually under a spawn here. These tests load the saved file instead.
+/// </summary>
+[TestClass]
+public class OutpostSceneTests
+{
+	const string ScenePath = "Assets/scenes/outpost.scene";
+
+	/// <summary>
+	/// OUT-SCENE-02 (draft): a player put at the saved scene's north spawn falls and comes
+	/// to rest on the scene's own "ground" object.
+	/// </summary>
+	[TestMethod]
+	public void OUT_SCENE_02_NorthSpawnLandsOnGround()
+	{
+		var scene = new Scene();
+		using ( scene.Push() )
+		{
+			Assert.IsTrue( EngineTestHelpers.LoadSavedScene( scene, ScenePath ),
+				$"{ScenePath} did not load." );
+
+			// Taken from the scene rather than from the layout file, so this checks where the
+			// spawn actually is in the saved scene.
+			var spawn = scene.GetAllObjects( true ).FirstOrDefault(
+				go => go.Name == "spawn.north" && go.Components.Get<SpawnPoint>() is not null );
+			Assert.IsNotNull( spawn,
+				$"{ScenePath} has no enabled \"spawn.north\" object with a SpawnPoint on it." );
+
+			var start = spawn.WorldPosition;
+			var pc = EngineTestHelpers.CreatePlayer( scene, start );
+
+			// Checked before the fall, not after: a player buried in geometry already counts as
+			// on the ground, which would make the landing check below pass while proving nothing.
+			Assert.IsFalse( EngineTestHelpers.BodyStartsInsideGeometry( pc ),
+				$"The north spawn at {start} starts the player inside geometry." );
+
+			// 60 ticks is 1.2s at this scene's 50Hz fixed update, far longer than a fall through
+			// the full height of the layout bounds, so failing here means nothing held it up.
+			Assert.IsTrue( EngineTestHelpers.TickUntilGrounded( scene, pc ),
+				$"The player never landed: it fell from {start} to {pc.GameObject.WorldPosition}." );
+
+			var under = EngineTestHelpers.TraceGroundUnder( pc );
+			Assert.IsTrue( under.Hit,
+				$"The player counts as grounded at {pc.GameObject.WorldPosition}, but nothing is under its feet." );
+			Assert.AreEqual( "ground", under.GameObject?.Name,
+				$"The player landed on \"{under.GameObject?.Name}\" at {pc.GameObject.WorldPosition}, not on the scene's ground." );
+		}
+	}
+}
